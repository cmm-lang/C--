// Возвращает найденную подстроку, если она есть
// Возвращает null, если не найдена
// В условиях: != string null - успех (найдено), string null - нет

const text2 = "Hello World!"

// Поиск подстроки
const result = strfind(text2, "World")   // result = "World"
say!(result)                           // World

// Если не найдено — null
const result2 = strfind(text2, "Mars")   // result2 = string null

// strfind_pos(x) возращает позицию
const pos = strfind_pos(text2, "Mars")
// mars не найден позиция = int null

// Проверка в условии (!= string null = найдено, string null = не найдено)
if (strfind(text2, "World") != string null) {
    say!("Найдено!")
}

const found = strfind(text2, "World")
if (found != string null) {
    say!("Найдено: ", found)
}

5. ИМПОРТ ФАЙЛОВ (import "file")
================================================================================

Теперь код можно разбивать на несколько файлов.

Синтаксис:
import "filename.cmm"

Где:
- filename.cmm — файл с кодом на C--
- Импортированный файл НЕ должен содержать main()
- main() должен быть только в главном файле
- не нужно указывать где находится файл с main, gcmt сам найдёт
- Расширение файлов: .cmm (C--)

ПРИМЕРЫ:

✅ Правильно (главный файл):

// main.cmm
import "math.cmm"
import "game.cmm"

fn main() {
    term_color("green")
    say!(add(10, 5))
    say!(player.hp)
}

✅ Правильно (библиотека):

// math.cmm
fn add(const a: int, const b: int) { return a + b }
fn sub(const a: int, const b: int) { return a - b }
const PI = 3.14159

❌ Неправильно (в библиотеке не должно быть main()):

// math.cmm  ❌ ОШИБКА!
fn add(const a: int, const b: int) { return a + b }
fn main() { say!("Hello") }   // ❌ main() только в main.cmm!


ПРАВИЛА:

1. main() должен быть ТОЛЬКО в main.cmm
2. Импортированные файлы (.cmm) НЕ должны содержать main()
3. Двойной импорт игнорируется (автоматическая защита)
4. Все функции из импортированного файла попадают в глобальное пространство
5. Если есть конфликт имён — ошибка компиляции
6. Чтобы скомпилировать файл с import надо:
gcmt -c main.cmm math.cmm program
gcmt -r program

7. ОБНОВЛЯЕМЫЙ ВЫВОД (say!_work())
================================================================================

Синтаксис:
say!_work("id", "текст") ИЛИ say_work("id", "текст")

Как это работает:
- Первый вызов с псевдонимом создаёт строку
- Последующие вызовы с тем же псевдонимом ЗАМЕНЯЮТ эту строку
- Каждый псевдоним — это отдельная строка

Пример:
fn main() {
    say!_work("progress", "Загрузка: 0%")
    say!_work("status", "Статус: Ожидание")
    say!_work("time", "Время: 0с")
    
    say!_work("progress", "Загрузка: 50%")
    say!_work("progress", "Загрузка: 100%")
}

Пример с потоками:
fn loader(var arg: void*) -> void* {
    for (i = 0, i <= 100, i++) {
        say!_work("progress", "Загрузка: ", i, "%")
        msleep(100)
    }
    say!_work("progress", "✅ Загрузка завершена!")
}

fn main() {
    thread("load", loader, void* null)
    thread_wait("load")
}

В других языках:
- Python: только через \r (одна строка)
- C: только через \r (одна строка)
- C--: ЛЮБОЕ КОЛИЧЕСТВО СТРОК!

11.1. МОДЫ
=============================
- УНИКАЛЬНАЯ ФИЧА
синтаксис:
value (x, mode) {
...
}
Вот все моды:
STANDART = стандартный
IGNORE_REGISTER = не регистрозависимый
STRFIND_MODE = как strfind
IGNORE_WHITESPACE = не зависимый от пробелов
Можно комбинировать с |:
IGNORE_REGISTER | STFIND_MODE

Пример standard, ignore_register, strfind_mode:
STANDARD:
value (x, STANDARD) {
"Привет": { say!("привет") }
default: { say!("не понял") }
}
Минусы:
Не работает: ПрИвЕт, привет, ПРИВЕТ, приВЕТ и тд
Нужно попасть точь в точь

IGNORE_REGISTER:
value (x, IGNORE_REGISTER) {
"привет" { say!("ПРИВЕТ") }
default: { say!("не понял") }
}
Плюсы:
Работает привет, Привет, ПрИвЕт, ПРИВЕТ, пРиВеТ, и тд
Минусы:
Нету

STRFIND_MODE:
value (x, STRFIND_MODE) {
"привет": { say!("привет") }
default: { say!("не понял") }
}
Плюсы:
Работает и привет, привет меня зовут Андрей
Минусы:
Не всегда нужен

12.1. ОБЫЧНЫЕ УКАЗАТЕЛИ (Decree)

Ключевое слово: ptr (поинт) или const ptr

Синтаксис:
    ptr *имя: (опционально: тип) = #переменная

Пример:
    const x = 10
   const ptr *p: *int = #x      // указатель *p указывает на x
    say!(*p)             // 10

Если так:
const x = 10
ptr *p = #x // ошибка! x неизменяемый а ptr изменяемый

Массив указателей:
var x = 10
var y = 20
ptr *p[2] = [#x, #y] // тип = [*int]


Особенности:
    - Оператор # — взятие адреса (как & в C)
    - Звёздочка обязательна в объявлении указателя
    - Звёздочка — разыменование
    - ptr — встроенный тип

Передача в функцию:
    fn change(ptr *p: *int) {
        *p = 42
    }

    fn main() {
        var a = 5
        ptr *p = #a
        change(*p)
        say!(a)   // 42
    }

12.2. УКАЗАТЕЛИ НА МАССИВЫ

Синтаксис:
    var arr = [1, 2, 3, 4, 5]
    ptr *p = #arr   // указатель на целый массив, тип = *[int]
    ptr *q = #arr[0] // указатель на элемент, тип = *int
Арифметика указателей (3 способа):

1. Смещение через +(int):
    say!(*(p + 1))       // второй элемент (индекс 1)

2. Доступ по индексу (как в C):
    say!(*p[i])          // доступ по индексу i

3. Сдвиг указателя (ptr++):
    p++
    say!(*p)             // второй элемент

Вот пример парсинга строки:
var name = "Андрей"
ptr *p = #name
var i = 1
while (*p != '\0') {
i++
p++
say!(i, " буква: ", *p)
}
Вывод:
2 буква: н
3 буква: д
4 буква: р
5 буква: е
6 буква: й
7 буква:   // тут \0


12.3. NULL УКАЗАТЕЛИ

Синтаксис:
    ptr *p = *type null      // указатель пуст
- если указатель пуст и ты пытаешься присвоить ему значение то будет ошибка
Пример:
    fn main() {
        ptr *p = *int null
        var x = 100
        p = #x            // направляем на x

        if (ptr == *int null) {
            say!("указатель пуст")
        }

        if (ptr != *int null) {
            say!("указатель не пуст")
        }
    }
Так нельзя:
const ptr *p = *int null
*p = 20

Особенности:
    - null (маленькими буквами)
    - Без ;
    - Без & (используем #)

14. THREADS
================================================================================

thread("id", Функция, Аргументы/void* null)
- создаёт поток
(возвращает: 0: ошибка, 1: всё нормально)

thread_wait("id");
- Ждёт конца Потока
(возвращает: void* то что вернул поток)

thread_kill("id");
- Убивает Поток
(возвращает: 0: ошибка, 1: всё нормально)

Пример:
fn worker(const arg: void*) -> void {
    const int_arg = (int)arg
    say!("Аргумент: ", int_arg)
    thread_kill("arg")
}

fn main() {
    thread("arg", worker, 42)
    thread_wait("arg")
    say!("program end")
}

Вывод:
Аргумент: 42
program end

ВАЖНО:
функция для потоков должна принимать все аргументы void* и должна возвращать void*
Иначе предупреждение или ошибка!!!


14.1. МЬЮТЕКСЫ
=============================
var x = mutex_create()
- создаёт мьютекс

mutex_init(x)
- инициализирует мьютекс

mutex_lock(x)
- блокирует мьютекс

mutex_unlock(x)
- разблокирует мьютекс

mutex_destroy(x)
- уничтожает мьютекс (обезательно для памяти)

- пример (на словах):
2 потока идут на одни данные
Ты создаёшь мьютекс (замок)
В функции для потока
Ты закрываешь mutex (mutex_lock)
И 2 поток не может зайти
И ему нужно будет ждать
Пока поток 1 не выполнит работу
А когда он выполнит работу
И откроет дверь (mutex_unlock)
А потом второй поток заходит и выполняет работу

ptr *x = (int)mem_alloc(int, 100)

for (var i = 0, i < 100, i++) {

ptr *x = (string)mem_alloc(string, 6)

x = "Hello"

ptr *p = (int)mem_alloc(int, 100)

22. LIST
================================================================================
- Весь список ключевых слов!

Функции:
... Ключевое слово для объявления в аргументах
list_t тип для list
list_init(list, тип) инициализирует list с любым типом возвращает void
list_next(list) переходит на слеюдущий элемент из аргументов
Сохраняет аргумент в list возвращает void
list_has_next(list) имеет ли list право на слеюдущий next возвращет bool

И вот пример:
fn print_all(first: string, ...) -> void {
    const arg: list_t
    list_init(arg, string)  // Указываем тип: string
    say!("Первый: ", first)
    
    while (list_has_next(arg)) {
        list_next(arg)
        say!("Следующий: ", arg)
    }
}

fn main() {
    print_all("Hello", "World", "!", "C--")
}
// Вывод:
// Первый: Hello
// Следующий: World
// Следующий: !
// Следующий: C--
