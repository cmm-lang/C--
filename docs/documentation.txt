0. ТИПЫ ДАННЫХ
================================================================================

char          - 1 символ, пишется с '' (например: 'A', '9', '#')
string[buf]   - много символов. Нужно указывать размер буфера, как в C.
Но можно сделать Динамическую строку:
var str[] = "text"
динамическую строку нельзя изменять по индексу
                Строку с буфером можно изменять по индексу: str[1] = 'A' (заменит второй символ)
int           - целое число (например: 10, -5, 0)
float         - число с плавающей точкой (например: 3.14, -2.5)
bool       - логическое значение (true или false)
type null             - пустое значение

Примеры:
    const letter = 'A'
const name[7]: string = "Andrey"   // string, буфер 7 байт (6 + \0)
name[0] = 'B'                       // теперь "Bndrey" ✅
const age = 10
const pi = 3.14159
const flag = true

// можно сравнивать строку по индексу:
if (name[0] == 'B') {
    say!("первая буква: B")
}

⚠️ Буфер строки указывается в БАЙТАХ, не в символах!
   ASCII:  1 символ = 1 байт
   UTF-8:  русская буква = 2 байта, эмодзи = 3-6 байт
   Всегда +1 байт под '\0'

   "Hello"   → нужно [6]
   "Andrey"  → нужно [7]
   "Андрей"  → нужно [13]

   Индексация строки — побайтовая!
   name[0] — первый БАЙТ, а не первый символ.
   Для UTF-8 это может быть часть символа.

   📌 В C-- 1.1 появится ustring — строка,
      которая считает СИМВОЛЫ, а не байты.

// Базовые функции
strlen(text)
- возвращает размер строки в байтах (без \0)
- для ASCII: 5 символов = 5 байт
- для UTF-8: русская буква = 2 байта, эмодзи = 3-6 байт
(возвращает: количество символов (int))

strupper(text)
- переводит строку в верхний регистр (возвращает: строку в uppercase)

strlower(text)
- переводит строку в нижний регистр (возвращает: строку в виде lowercase)

strtrim(text)
- убирает пробелы по краям
(возвращает: строку)

5. РАБОТА С ФАЙЛАМИ
================================================================================

create_file("file.txt") {        - создать файл и записать
    fsay!("текст")
    fsay!(переменная)
}
(возвращает: 0: файл не создан, 1: всё нормально)

append_file("file.txt") {        - добавить в конец файла
    fsay!("текст")
}
(возвращает 0: ошибка, 1: всё нормально)

write_file("file.txt") {         - перезаписать файл
    fsay!("новый текст")
}
(возвращает: 0: ошибка, 1: всё нормально)

const content[] = read_file("file.txt")  - прочитать весь файл
(возвращает: == string null: ошибка, != string null: всё нормально)

if (!file_exists("file.txt")) {                - проверка существования файла
    // файла нет
}
(возвращает: bool, true найдено false не найдено)

⚠️ ВАЖНО: Всегда проверяйте результат на ошибку (0)!
⚠️ ВАЖНО: create_file, append_file, write_file, file_exists принимает только строковой литерал
Пример:
    // Создание и запись
    const err = create_file("notes.txt") {
        fsay!("Мои заметки")
        fsay!("Список дел:")
        fsay!("1. Сделать C--")
        fsay!("2. Выучить C")
    }
    
    // Проверка ошибки
    if (err == 0) {
        fatal("Ошибка создания файла!")
    }
    
    // Проверка существования
    if (!file_exists("notes.txt")) {
        fatal("Файл не найден!")
    }
    
    // Чтение
    const content[] = read_file("notes.txt")
    if (content == string null) {
        fatal("Ошибка чтения!")
    }
    say!("Содержимое: ", content)
    
    // Добавление в конец
    append_file("notes.txt") {
        fsay!("\n3. Сделать компилятор")
        fsay!("4. Запушить на GitHub")
    }
    
    // Перезапись
    write_file("notes.txt") {
        fsay!("НОВЫЙ СПИСОК:")
        fsay!("1. C-- 1.0")
        fsay!("2. C-- 1.1")
    }

6. АРИФМЕТИКА И СРАВНЕНИЕ
================================================================================

Арифметические операторы:
    +   - сложение        (5 + 3 → 8)
    -   - вычитание       (5 - 3 → 2)
    *   - умножение       (5 * 3 → 15)
    /   - деление         (6 / 3 → 2)
    %  - остаток          (10 % 3 → 1)
    ++  - инкремент       (x++ → x + 1)  // только для var
    --  - декремент       (x-- → x - 1)  // только для var

Приореты:
() - скобки                    ((5 + 5) * (5 + 5) = 100)

Операторы сравнения:
    ==  - равно           (5 == 5 → true)
    !=  - не равно        (5 != 3 → true)
    >   - больше          (5 > 3 → true)
    <   - меньше          (5 < 3 → false)
    >=  - больше или равно (5 >= 5 → true)
    <=  - меньше или равно (5 <= 3 → false)

Логические операторы:
    &&   - И              (true && false → false)
    ||   - ИЛИ            (true || false → true)
    !    - НЕ             (!true → false)

Пример:
    var counter = 0
    counter++
    say!(counter)   // 1

10. ФУНКЦИИ
================================================================================

fn name(квалификатор params: type) -> returning type { }             - объявление функции
return value                    - возврат значения
name(args)                      - вызов функции

Квалификаторы:
const, var
v1.2:
ptr, const ptr

Пример:
    fn add(const x: int, const y: int) -> int {
        return x + y
    }
    
    fn main() {
        const sum = add(4, 4)
        say!(sum)  // 8
    }

Важно:
- в аргументы можно передавать void*
fn foo(var x: void*) -> void*
Значит тип может быть любой

функции в аргументы можно передавать обязательно с типом fn и с const/var (всё равно функции нельзя не как менять):
fn function() -> void {
say!("Hello")
}

fn foo(const x: fn() -> void) -> void {
x()
}

fn main() {
foo(function)
}
Если там аргументы то надо писать аргументы например:
fn add(const x: int, const y: int) {
return x + y
}

fn foo(const x: fn(int, int), const y: int, const z: int) {
return x(y, z)
}

fn main() {
foo(add, 10, 10) // 20
}
Сложные задачи:
fn foo(const callback: fn(int) -> int) {
    // Здесь callback — это функция
    // Мы её вызываем с аргументом
    const result = callback(42)
    say!(result)
}

fn double(const x: int) -> int {
    return x * 2
}

fn main() {
    // ✅ Правильно: передаём функцию без вызова
    foo(double)  // double без скобок!
    
    // ❌ Неправильно: передаём результат вызова
    foo(double(5))  // Ошибка! double(5) — это int, а не fn
}

Рекурсия тоже разрешена:
fn foo() {
say!("Hello")
for (var i = 0, i < 5, i++) {
foo()
}
 }
fn main() {
foo()
}
Вывод: Hello очень много раз (Stack overflow) надо в цикле в main:
fn main() {
for (var i = 0, i < 5, i++) {
foo()
}
 }
~ 25 раз выведется Hello

15. Компилятор: GCMT
================================================================================
Полное Название: GNU C-- Translator

Установка:
pkg install gcmt
Либо ваш установщик

gcmt -c file.cmm program
- компилирует файл

Все Моды:
-c: скомпилировать
-r: запустить
-e: вывести все ошибки
-warn: не выдавать предупреждения
-cc: выдать код сгенерированный C код

Примеры:
gcmt -cc file.cmm file.c // сохраняем в file.c
gcmt -warn -c file.cmm program // предупреждений нету
gcmt -e file.cmm // выдаёт все ошибки без компиляции
gcmt -r program // запускает

NODE_MINUS,
NODE_STAR,
NODE_WAND
}
struct AST {
const node: node
const value: string[]
const next: AST
}

const ASTNode = AST{NODE_PLUS, "+", AST{NODE_MINUS, "-", AST{NODE_STAR, "*", AST{NODE_WAND, "/", AST null}}}}

say!(ASTNode.value) // "+"
say!(ASTNode.next.value) // "-"
say!(ASTNode.next.next.value) // "*"
say!(ASTNode.next.next.next.value) // "/"

20. NULL Аргументы
================================================================================
fn greet(const name: string) -> string {
    if (name == string null) {
        return "Hello, guest"
    }
    else {
        const greeting[] = "Hello, ", name
        return greeting
}

fn main() {
    const x = greet(string null)
    say!(x)
    // вывод: Hello, guest
}

// Программа: список дел

fn main() {
    var task[500] = string null
    var choice = int null
    
    while (true) {
        say!("\n===== СПИСОК ДЕЛ =====")
        say!("1. Показать задачи")
        say!("2. Добавить задачу")
        say!("3. Очистить всё")
        say!("4. Выход")
        input(choice, int)
        
        if (choice == 1) {
            const tasks[] = read_file("tasks.txt")
            if (tasks == string null) {
                say!("Ошибка при чтении файла!")
            } else {
                if (!file_exists ("tasks.txt")) {
                    say!("Задач нет")
                } else {
                    say!("Задачи: \n", tasks)
                }
            }
            press()
            cls()
            
        } else if (choice == 2) {
            say!("Введи задачу")
            input(task, string)
          const c =  append_file("tasks.txt") {
                fsay!(task)
            }
            if (c == 0) {
                say!("Ошибка при сохранении задачи!")
            } else {
                say!("Добавлено!")
            }
            ssleep(1)
            cls()
            
        } else if (choice == 3) {
         const c =   write_file("tasks.txt") {
                fsay!("")
            }
            if (c == 0) {
                say!("Ошибка при очистке файла!")
            } else {
                say!("Всё очищено!")
            }
            ssleep(1)
            cls()
            
        } else if (choice == 4) {
            say!("До свидания, командир!")
            break
            
        } else {
            say!("Неизвестная команда")
            ssleep(1)
            cls()
        }
    }
}

| Команда       | Назначение              | Синтаксис                                    |
|---------------|-------------------------|----------------------------------------------|
| const           | Константа               | const x = 10                                   |
| var           | Переменная              | var x = 10                                   |
| say!()        | Вывод с переносом       | say!("текст")                                |
| say()         | Вывод без переноса      | say("текст")                                 |
| input()       | Ввод с проверкой типа   | input(var, int)                              |
| cls()         | Очистка экрана          | cls()                                        |
| press()       | Ожидание Enter          | press()                                      |
| ssleep()      | Пауза (сек)             | ssleep(5)                                    |
| msleep()      | Пауза (мс)              | msleep(500)                                  |
| create_file()  | Создать файл            | create_file("file.txt") { fsay!() }           |
| append_file()  | Добавить в конец        | append_file("file.txt") { fsay!() }           |
| write_file()   | Перезаписать            | write_file("file.txt") { fsay!() }            |
| read_file()    | Прочитать               | const data = read_file("file.txt")              |
| file_exists("file")         | Проверка существования  | if (file_exists("file.txt")) { ... }                       |
| if            | Условие                 | if (x > 5) { ... }                           |
| while         | Цикл                    | while (x < 10) { ... }                       |
| for           | Цикл со счётчиком       | for (var i = 0, i < 10, i++) { ... }             |
| break         | Выход из цикла          | break                                        |
| continue      | Пропуск итерации        | continue                                     |
| //            | Комментарий             | // текст                                     |
| fatal()       | Фатальная ошибка        | fatal("сообщение")                           |
| fn            | Объявление функции      | fn name() -> void* { ... }                            |
| return        | Возврат из функции      | return value                                 |
| strlen()      | Длина строки в байтах без \0           | strlen("Hello") → 5      |
| strupper()    | Верхний регистр         | strupper("hi") → "HI"    |
| strlower()    | Нижний регистр          | strlower("HI") → "hi"    |
| strtrim()     | Удалить пробелы         | strtrim("  hi  ") → "hi" |
| struct name | структура данных | struct Human { const name[500]: string } |
| enum name | перечисления | enum TokenType { TOKEN_LET } |
