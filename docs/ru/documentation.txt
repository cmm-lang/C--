================================================================================

char          - 1 символ, пишется с '' (например: 'A', '9', '#')
string[buf]   - много символов. Нужно указывать размер буфера, как в C.
Но можно сделать Динамическую строку:
var str[] = "text"
её можно менять по индексу
                Строку можно изменять по индексу: str[1] = 'A' (заменит второй символ)
int           - целое число (например: 10, -5, 0)
float         - число с плавающей точкой (например: 3.14, -2.5)
bool       - логическое значение (true или false)
type null             - пустое значение

Примеры:
    const letter = 'A'
const name[7]: string = "Andrey"   // string, буфер 7 байт (6 + \0)
name[0] = 'B'                       // теперь "Bndrey" ✅
const age = 10
const pi = 3.14159
const flag = true

// можно сравнивать строку по индексу:
if (name[0] == 'B') {
    say!("первая буква: B")
}

⚠️ Буфер строки указывается в БАЙТАХ, не в символах!
   ASCII:  1 символ = 1 байт
   UTF-8:  русская буква = 2 байта, эмодзи = 3-6 байт
   Всегда +1 байт под '\0'

   "Hello"   → нужно [6]
   "Andrey"  → нужно [7]
   "Андрей"  → нужно [13]

   Индексация строки — побайтовая!
   name[0] — первый БАЙТ, а не первый символ.
   Для UTF-8 это может быть часть символа.

   📌 В C-- 1.1 появится ustring — строка,
      которая считает СИМВОЛЫ, а не байты.

- возвращает размер строки в байтах (без \0)

- переводит строку в нижний регистр (возвращает: строку в виде lowercase)

strtrim(text)
- убирает пробелы по краям
(возвращает: строку)

create_file("file.txt") {        - создать файл и записать
    fsay!("текст")
    fsay!(переменная)
}
(возвращает: 0: файл не создан, 1: всё нормально)

append_file("file.txt") {        - добавить в конец файла
    fsay!("текст")
}
(возвращает 0: ошибка, 1: всё нормально)

write_file("file.txt") {         - перезаписать файл
    fsay!("новый текст")
}
(возвращает: 0: ошибка, 1: всё нормально)

const content[] = read_file("file.txt")  - прочитать весь файл
(возвращает: == string null: ошибка, != string null: всё нормально)

if (!file_exists("file.txt")) {                - проверка существования файла
    // файла нет
}
(возвращает: bool, true найдено false не найдено)

Как передавать файлы в функции?:
fn add_to_file(file: string, msg: string) {
append_file(file) {
fsay!(msg)
}
 }

fn main() {
create_file("file") {

}
add_to_file("file", "работает")
}

⚠️ ВАЖНО: Всегда проверяйте результат на ошибку (0)!

Пример:
    // Создание и запись
    const err = create_file("notes.txt") {
        fsay!("Мои заметки")
        fsay!("Список дел:")
        fsay!("1. Сделать C--")
        fsay!("2. Выучить C")
    }
    
    // Проверка ошибки
    if (err == 0) {
        fatal("Ошибка создания файла!")

    // Проверка существования
    if (!file_exists("notes.txt")) {
        fatal("Файл не найден!")
    }
    
    // Чтение
    const content[] = read_file("notes.txt")
    if (content == string null) {

    }

    // Добавление в конец

6. АРИФМЕТИКА И СРАВНЕНИЕ
================================================================================

Арифметические операторы:
    +   - сложение        (5 + 3 → 8)
    -   - вычитание       (5 - 3 → 2)
    *   - умножение       (5 * 3 → 15)
    /   - деление         (6 / 3 → 2)
    %  - остаток          (10 % 3 → 1)
    ++  - инкремент       (x++ → x + 1)  // только для var
    --  - декремент       (x-- → x - 1)  // только для var

Приореты:
() - скобки                    ((5 + 5) * (5 + 5) = 100)

Операторы сравнения:
    ==  - равно           (5 == 5 → true)
    !=  - не равно        (5 != 3 → true)
    >   - больше          (5 > 3 → true)
    <   - меньше          (5 < 3 → false)
    >=  - больше или равно (5 >= 5 → true)
    <=  - меньше или равно (5 <= 3 → false)

Логические операторы:
    &&   - И              (true && false → false)
    ||   - ИЛИ            (true || false → true)
    !    - НЕ             (!true → false)

Пример:
    var counter = 0
    counter++
    say!(counter)   // 1

10. ФУНКЦИИ
================================================================================

fn name(квалификатор params: type) -> returning type { }             - объявление функции
return value                    - возврат значения
name(args)                      - вызов функции

Квалификаторы:
const, var
v1.2:
ptr, const ptr

Пример:
    fn add(const x: int, const y: int) -> int {
        return x + y
    }
    
    fn main() {
        const sum = add(4, 4)
        say!(sum)  // 8
    }

Важно:
- в аргументы можно передавать void*
fn foo(var x: void*) -> void*
Значит тип может быть любой

функции в аргументы можно передавать обязательно с типом fn и с const/var (всё равно функции нельзя не как менять):
fn function() -> void {
say!("Hello")
}

fn foo(const x: fn() -> void) -> void {
x()
}

fn main() {
foo(function)
}
Если там аргументы то надо писать аргументы например:
fn add(const x: int, const y: int) {
return x + y
}

fn foo(const x: fn(int, int), const y: int, const z: int) {
return x(y, z)
}

fn main() {
foo(add, 10, 10) // 20
}
Сложные задачи:
fn foo(const callback: fn(int) -> int) {
    // Здесь callback — это функция
    // Мы её вызываем с аргументом
    const result = callback(42)
    say!(result)
}

fn double(const x: int) -> int {
    return x * 2
}

fn main() {
    // ✅ Правильно: передаём функцию без вызова
    foo(double)  // double без скобок!
    
    // ❌ Неправильно: передаём результат вызова
    foo(double(5))  // Ошибка! double(5) — это int, а не fn
}

Рекурсия тоже разрешена:
fn foo() {
say!("Hello")
for (var i = 0, i < 5, i++) {
foo()
}
 }
fn main() {
foo()
}
Вывод: Hello очень много раз (Stack overflow) надо в цикле в main:
fn main() {
for (var i = 0, i < 5, i++) {
foo()
}
 }
~ 25 раз выведется Hello

15. Компилятор: GCMT
================================================================================
Полное Название: GNU C-- Translator

Установка:
pkg install gcmt
Либо ваш установщик

gcmt -c file.cmm program
- компилирует файл

Все Моды:
-c: скомпилировать
-r: запустить
-e: вывести все ошибки
-warn: не выдавать предупреждения
-eue (error unexpected): не выдавать ошибки
-cc: выдать код сгенерированный C код

Примеры:
gcmt -cc file.cmm file.c // сохраняем в file.c
gcmt -eue -c file.cmm program // если там ошибка то он не выдаёт её
gcmt -warn -c file.cmm program // предупреждений нету
gcmt -e file.cmm // выдаёт все ошибки без компиляции
gcmt -r program // запускает

5. ИМПОРТ ФАЙЛОВ (import "file")
================================================================================

Теперь код можно разбивать на несколько файлов.

Синтаксис:
import "filename.cmm"

Где:
- filename.cmm — файл с кодом на C--
- Импортированный файл НЕ должен содержать main()
- main() должен быть только в главном файле
- не нужно указывать где находится файл с main, gcmt сам найдёт
- Расширение файлов: .cmm (C--)

ПРИМЕРЫ:

✅ Правильно (главный файл):

// main.cmm
import "math.cmm"
import "game.cmm"

fn main() {
    term_color("green")
    say!(add(10, 5))
    say!(player.hp)
}

✅ Правильно (библиотека):

// math.cmm
fn add(const a: int, const b: int) { return a + b }
fn sub(const a: int, const b: int) { return a - b }
const PI = 3.14159

❌ Неправильно (в библиотеке не должно быть main()):

// math.cmm  ❌ ОШИБКА!
fn add(const a: int, const b: int) { return a + b }
fn main() { say!("Hello") }   // ❌ main() только в main.cmm!


ПРАВИЛА:

1. main() должен быть ТОЛЬКО в main.cmm
2. Импортированные файлы (.cmm) НЕ должны содержать main()
3. Двойной импорт игнорируется (автоматическая защита)
4. Все функции из импортированного файла попадают в глобальное пространство
5. Если есть конфликт имён — ошибка компиляции
6. Чтобы скомпилировать файл с import надо:
gcmt -c main.cmm math.cmm program
gcmt -r program

7. ОБНОВЛЯЕМЫЙ ВЫВОД (say!_work())
================================================================================

Синтаксис:
say!_work("id", "текст") ИЛИ say_work("id", "текст")

Как это работает:
- Первый вызов с псевдонимом создаёт строку

fn main() {

    say!_work("progress", "Загрузка: 50%")
    say!_work("progress", "Загрузка: 100%")
}

fn loader(var arg: void*) -> void* {

        msleep(100)
    }
    say!_work("progress", "✅ Загрузка завершена!")
}

fn main() {
    thread("load", loader, void* null)
    thread_wait("load")

13.1. ОБЫЧНЫЕ УКАЗАТЕЛИ (Decree)

Ключевое слово: ptr (поинт) или const ptr

Синтаксис:
    ptr *имя: (опционально: тип) = #переменная

Пример:
    const x = 10
   const ptr *p: *int = #x      // указатель *p указывает на x
    say!(*p)             // 10

Если так:
const x = 10
ptr *p = #x // ошибка! x неизменяемый а ptr изменяемый

Массив указателей:
var x = 10
var y = 20
ptr *p[2] = [#x, #y] // тип = [*int]


Особенности:
    - Оператор # — взятие адреса (как & в C)
    - Звёздочка обязательна в объявлении указателя
    - Звёздочка — разыменование
    - ptr — встроенный тип

Передача в функцию:
    fn change(ptr *p: *int) {
        *p = 42
    }

    fn main() {
        var a = 5
        ptr *p = #a
        change(*p)
        say!(a)   // 42
    }

13.2. УКАЗАТЕЛИ НА МАССИВЫ

Синтаксис:
    var arr = [1, 2, 3, 4, 5]
    ptr *p = #arr   // указатель на целый массив, тип = *[int]
    ptr *q = #arr[0] // указатель на элемент, тип = *int
Арифметика указателей (3 способа):

1. Смещение через +(int):
    say!(*(p + 1))       // второй элемент (индекс 1)

2. Доступ по индексу (как в C):
    say!(*p[i])          // доступ по индексу i

3. Сдвиг указателя (ptr++):
    p++
    say!(*p)             // второй элемент

Вот пример парсинга строки:
var name = "Андрей"
ptr *p = #name
var i = 1
while (*p != '\0') {
i++
p++
say!(i, " буква: ", *p)
}
Вывод:
2 буква: н
3 буква: д
4 буква: р
5 буква: е
6 буква: й
7 буква:   // тут \0

Синтаксис:
    ptr *p = *type null      // указатель пуст
- если указатель пуст и ты пытаешься присвоить ему значение то будет ошибка
Пример:
    fn main() {
        ptr *p = *int null
        var x = 100
        p = #x            // направляем на x

        if (ptr == *int null) {
            say!("указатель пуст")
        }

        if (ptr != *int null) {
            say!("указатель не пуст")
        }
    }
Так нельзя:
const ptr *p = *int null
*p = 20

Особенности:
    - null (маленькими буквами)
    - Без ;
    - Без & (используем #)

15. THREADS
================================================================================

thread("id", Функция, Аргументы/void* null)
- создаёт поток
(возвращает: 0: ошибка, 1: всё нормально)

thread_wait("id");
- Ждёт конца Потока
(возвращает: void* то что вернул поток)

thread_kill("id");
- Убивает Поток
(возвращает: 0: ошибка, 1: всё нормально)

Пример:
fn worker(const arg: void*) -> void {
    const int_arg = (int)arg
    say!("Аргумент: ", int_arg)
    thread_kill("arg")
}

fn main() {
    thread("arg", worker, 42)
    thread_wait("arg")
    say!("program end")
}

Вывод:
Аргумент: 42
program end

ВАЖНО:
функция для потоков должна принимать все аргументы void* и должна возвращать void*
Иначе предупреждение не ошибка!!!

var x = mutex_create()
- создаёт мьютекс

mutex_init(x)
- инициализирует мьютекс

mutex_lock(x)
- блокирует мьютекс

mutex_unlock(x)
- разблокирует мьютекс

mutex_destroy(x)
- уничтожает мьютекс (обезательно для памяти)

- пример (на словах):
2 потока идут на одни данные
Ты создаёшь мьютекс (замок)
В функции для потока
Ты закрываешь mutex (mutex_lock)
И 2 поток не может зайти

Пока поток 1 не выполнит работу
А когда он выполнит работу
И откроет дверь (mutex_unlock)
А потом второй поток заходит и выполняет работу

25. LIST

Функции:
... Ключевое слово для объявления в аргументах
list_t тип для list

fn print_all(first: void*, ...) -> void {
    const arg: list_t
    list_init(arg, void*)  // Указываем тип: void*
    say!("Первый: ", (string)first)
    
    while (list_has_next(arg)) {
        list_next(arg)
        say!("Следующий: ", (string)arg)
    }
}

fn main() {
    print_all("Hello", "World", "!", "C--")
}
// Вывод:
// Первый: Hello
// Следующий: World
// Следующий: !
// Следующий: C--
